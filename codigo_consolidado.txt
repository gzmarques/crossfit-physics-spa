  5 | export function parseClockTime(str: string | number): number {
  6 |   if (str === undefined || str === null) return -1;
  7 |   const cleanStr = str.toString().trim();
  8 |   if (cleanStr === "") return -1;
  9 | 
 10 |   if (!cleanStr.includes(':')) {
 11 |     const val = Number(cleanStr);
 12 |     return (!isNaN(val) && val >= 0) ? val : -1;
 13 |   }
 14 | 
 15 |   const parts = cleanStr.split(':').map(p => Number(p.trim()));
 16 |   if (parts.some(p => isNaN(p) || p < 0)) return -1;
 17 | 
 18 |   if (parts.length === 2) {
 19 |     return (parts[0] * 60) + parts[1];
 20 |   } else if (parts.length === 3) {
 21 |     return (parts[0] * 3600) + (parts[1] * 60) + parts[2];
 22 |   }
 23 |   return -1;

656 | 
657 |       const tut_default = deltaT > 0 ? deltaT : 3.0; 
658 |       if (['pegboard', 'hs_walk', 'wall_walk', 'hs_incline_up', 'hs_incline_down'].includes(cfg.categoria)) { 
659 |           E_isom = (tMech > 0 ? tMech : atleta.peso * G * 0.50) * tut_default * 0.80; 
660 |       } else if (['ring_muscle_up', 'strict_rmu'].includes(movId)) { 
661 |           E_isom = tMech * tut_default * 0.40; 
662 |       } else if (['overhead_squat', 'turkish_get_up'].includes(movId) || cfg.categoria === 'core_t2b' || cfg.categoria === 'core_ghd') { 
663 |           E_isom = tMech * tut_default * 0.30; 
664 |       } else if (cfg.categoria.includes('friccao_horizontal')) { 
665 |           E_isom = tMech * tut_default * 0.40; 
666 |       }
667 | 
668 |       // Refatoração 5 (Cont.): Respeito absoluto à termodinâmica gravitacional
669 |       const W_conc = tMech; 
670 |       const W_exc = tMech * W_exc_ratio;
671 |       
672 |       const E_met_conc = W_conc / (Math.max(0.01, eta_conc) * 4184.0);
673 |       
674 |       // Integra a dissipação GRF no custo excêntrico (se for um movimento de LPO)
675 |       const custoExtraGRF = (cfg.categoria.includes('lpo_') || cfg.categoria.includes('db_') || cfg.categoria === 'deadlift') ? dissipacaoExcentricaGRF : 0;
676 |       const E_met_exc = (W_exc + custoExtraGRF) / (eta_exc * 4184.0);
677 |       
678 |       const E_met_isom = E_isom / 4184.0;
679 | 
680 |       // --- NOVA REFATORAÇÃO 4: Balanço W' (W-prime) e Potência Crítica (CP) ---
681 |       // A lógica e os parâmetros iniciais são validados.
682 |       const CP = atleta.peso * (atleta.nivelTecnico === 'avancado' ? 4.5 : atleta.nivelTecnico === 'intermediario' ? 3.0 : 2.0);
683 |       
684 |       // Processamento padrão aplicado para transição direta ao balanço térmico.
685 |       const tut_metabolico = deltaT > 0 ? deltaT : 3.0;
686 |       const potencia_execucao = (W_conc + W_exc + E_isom) / tut_metabolico;
687 |       
688 |       let W_prime_expenditure = 0;
689 |       if (potencia_execucao > CP) {
690 |           W_prime_expenditure = (potencia_execucao - CP) * tut_metabolico;
691 |       }
692 |       
693 |       const custoExtraMetabolicoWPrime = W_prime_expenditure / 4184.0;
694 | 
695 |       tMetWork = E_met_conc + E_met_exc + E_met_isom + (custoExtraMetabolicoWPrime * 1.25);
696 |       tMetWorkConcIsom = E_met_conc + E_met_isom + (custoExtraMetabolicoWPrime * 1.25);
697 |   }
698 | 
699 |   return { 
700 |       trabMech: tMech * safeReps, 
701 |       trabMetabolicoWork: tMetWork * safeReps, 
702 |       trabMetabolicoConcIsom: tMetWorkConcIsom * safeReps, 
703 |       infoExtraLog: exL, 
704 |       isErgo: isErgo, 
705 |       isCalorieErgo: isCalorieErgo,
706 |       ergTime: sErgo * safeReps
707 |   };
708 | }
