  5 | export function parseClockTime(str: string | number): number {
  6 |   if (str === undefined || str === null) return -1;
  7 |   const cleanStr = str.toString().trim();
  8 |   if (cleanStr === "") return -1;
  9 | 
 10 |   if (!cleanStr.includes(':')) {
 11 |     const val = Number(cleanStr);
 12 |     return (!isNaN(val) && val >= 0) ? val : -1;
 13 |   }
 14 | 
 15 |   const parts = cleanStr.split(':').map(p => Number(p.trim()));
 16 |   if (parts.some(p => isNaN(p) || p < 0)) return -1;
 17 | 
 18 |   if (parts.length === 2) {
 19 |     return (parts[0] * 60) + parts[1];
 20 |   } else if (parts.length === 3) {
 21 |     return (parts[0] * 3600) + (parts[1] * 60) + parts[2];
 22 |   }
 23 |   return -1;
